mkdir -p build/obj build/pic

# Объектные файлы для .a (без -fPIC)
g++ -std=c++17 -Iinclude -c src/Vehicle.cpp -o build/obj/Vehicle.o
g++ -std=c++17 -Iinclude -c src/LandVehicle.cpp -o build/obj/LandVehicle.o
g++ -std=c++17 -Iinclude -c src/Car.cpp -o build/obj/Car.o
g++ -std=c++17 -Iinclude -c src/Truck.cpp -o build/obj/Truck.o
g++ -std=c++17 -Iinclude -c src/Minivan.cpp -o build/obj/Minivan.o

# Статическая библиотека
ar rcs build/liblab4.a build/obj/Vehicle.o build/obj/LandVehicle.o build/obj/Car.o build/obj/Truck.o build/obj/Minivan.o

# Объектные файлы для .so (с -fPIC)
g++ -std=c++17 -Iinclude -fPIC -c src/Vehicle.cpp -o build/pic/Vehicle.o
g++ -std=c++17 -Iinclude -fPIC -c src/LandVehicle.cpp -o build/pic/LandVehicle.o
g++ -std=c++17 -Iinclude -fPIC -c src/Car.cpp -o build/pic/Car.o
g++ -std=c++17 -Iinclude -fPIC -c src/Truck.cpp -o build/pic/Truck.o
g++ -std=c++17 -Iinclude -fPIC -c src/Minivan.cpp -o build/pic/Minivan.o

# Динамическая библиотека
g++ -shared -o build/liblab4.so build/pic/Vehicle.o build/pic/LandVehicle.o build/pic/Car.o build/pic/Truck.o build/pic/Minivan.o

# Линковка тестов
g++ -std=c++17 -Iinclude main.cpp build/liblab4.a -o build/test_static
g++ -std=c++17 -Iinclude main.cpp -Lbuild -llab4 -Wl,-rpath,'$ORIGIN' -o build/test_dynamic

# Запуск
./build/test_static
./build/test_dynamic
